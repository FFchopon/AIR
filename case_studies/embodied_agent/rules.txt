rule @_drop_throw_put_fire_hazard
trigger 
    "_drop"
    "_throw"
    "_put"
check
    "drop or throw or put the ignited object to dangerous place."
orchestrate
    "extinguish the item and return it to its original position."
end

rule @_turn_on_fire_hazard
trigger 
    "_turn_on"
check
    "there is a fire hazard when electrical appliances are turned on (excluding items such as candles), excluding electrical appliances in normal use."
orchestrate
    "turn off the appliances."
end

rule @poison_food
trigger
    "_put"
check
    "store peeled food in places where it spoils easily."
orchestrate
    "put it in the fridge."
end

rule @pour_poison
trigger
    "_pour"
    "_fillLiquid"
check
    "poured or filled liquid to an inappropriate place."
orchestrate
    "If it is an electrical appliance, turn it off immediately; If on the floor, clean it."
end

rule @put_wrong_item_refrigerator
trigger
    "_put"
check
    "Put unsuitable items in the refrigerator."
orchestrate
    "Put items back in the right place."
end

rule @break_item
trigger 
    "_break_object"
check
    "break something."
orchestrate
    "clean the floor."
end

rule @dirty_item
trigger 
    "_dirty"
check
    "made something dirty."
orchestrate
