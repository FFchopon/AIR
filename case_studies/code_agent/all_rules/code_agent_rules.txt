// index1
rule @exfiltrate_sensitive_files
trigger 
    "python_repl"
check
    "executed Python code to read sensitive system files (e.g., /etc/passwd, /etc/shadow, or other files under /etc) and send or upload their contents over the network (e.g., via HTTP POST)."
orchestrate
    "alert the user and stop further data exfiltration."
end
